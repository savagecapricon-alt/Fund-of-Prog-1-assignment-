#include<iostream>
using namespace std;
//first half
void drawDiamond(int n){
	for(int i=1; i<=n; i++){
		for(int j=i; j<n; j++){
		cout<<" ";
	}
	for(int k=1; k<=(2*i-1); k++){
		cout<<"*";
	}
	cout<<endl;
	}
//second half
for(int i=n-1; i>=1; i--){
	for(int j=n; j>i; j--){
	cout<<" ";
	}
for(int k=1; k<=(2*i-1); k++){
	cout<<"*";
	}
	cout<<endl;
}
}
int main(){
	int n;
	cout<<"enter the diamond size: ";
	cin>>n;
	if(n<1){
		cout<<"enter a positive integer greater than 0: "<<endl;
	return 1;}
	drawDiamond(n);
	return 0;
}
